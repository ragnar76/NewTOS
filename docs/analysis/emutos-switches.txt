# EmuTOS-Buildoptionen – Kurzbeschreibung nach references/emutos/include/config.h.
# Bei Ein/Aus-Schaltern: 1 = einkompiliert/aktiviert, 0 = deaktiviert (Ausnahmen stehen in der Zeile).
# Hardwarefunktionen benötigen zusätzlich passende Hardware; dies sind keine Laufzeitschalter.
# Größen und Adressen sind Zahlenwerte. Die vorhandenen Werte bleiben unverändert.
# Markierungen: [GRAFIK] = Grafik/VDI/Videohardware; [DESKTOP] = Desktop/AES/GUI-Bedienung einschließlich Diagnose und interner Optionen.

CONF_ATARI_HARDWARE 1                             # Atari-Hardware als Plattformbasis; 1 setzt Atari-Chip-Unterstützung als Standard, 0 eine Plattform ohne diese Chips.
CONF_COLDFIRE_TIMER_C 0                           # Timer-C-Nachbildung durch interne ColdFire-Timer; 1 aktiviert sie, hier ausgeschaltet.
CONF_DEBUG_AES_STACK 0                            # [DESKTOP] Diagnose des AES-Stacks (GUI-System); 1 protokolliert den Verbrauch zur Größenabschätzung, hier aus.
CONF_DEBUG_DESK_STACK 0                           # [DESKTOP] Diagnose des Desktop-Stacks; 1 überwacht den Stackverbrauch, hier aus.
CONF_DETECT_FIRST_BOOT_WITHOUT_MEMCONF 0          # Erkennung des ersten Starts ohne RAM-Konfiguration; 1 erlaubt sie auch bei bereits initialisiertem RAM, hier aus.
CONF_LOGSEC_SIZE 512                              # Vorgabe für die maximale logische GEMDOS-Sektorgröße in Byte; hier 512, beim Start erkannte größere Sektoren erhöhen sie.
CONF_PREFER_STRAM_DISK_BUFFERS 1                  # [DESKTOP] Desktop-Datenträgerpuffer bevorzugt im ST-RAM; 1 erleichtert direkte Floppy-/ACSI-DMA-Transfers.
CONF_SERIAL_CONSOLE 0                             # Serielle Konsole; 1 gibt zusätzlich seriell aus und nimmt Konsoleneingaben ausschließlich seriell an, hier aus.
CONF_SERIAL_CONSOLE_ANSI 0                        # Steuerzeichen der seriellen Konsole; 1 verwendet ANSI, 0 Atari-VT52; nur bei serieller Konsole relevant.
CONF_SERIAL_CONSOLE_POLLING_MODE 0                # Serielle Konsoleneingabe; 1 fragt Daten aktiv ab (Polling), 0 verwendet Interrupts.
CONF_SERIAL_IKBD 0                                # Atari-Tastaturprozessor (Tastatur/Maus/Joystick) am seriellen Port; 1 erlaubt diesen Anschluss, hier aus.
CONF_STRAM_SIZE 0                                 # Feste ST-RAM-Größe in Byte; 0 erkennt die Größe automatisch.
CONF_TTRAM_SIZE 0                                 # Feste TT-RAM-Größe in Byte; 0 erkennt die Größe automatisch.
CONF_VRAM_ADDRESS 0                               # [GRAFIK] Feste Bildspeicheradresse für eigene Grafikhardware; 0 reserviert den Bildspeicher regulär im ST-RAM.
CONF_WITH_1FAT_SUPPORT 0                          # Datenträger mit nur einer FAT (Dateizuordnungstabelle); 1 unterstützt sie, 0 nimmt aus Atari-TOS-Kompatibilität zwei FATs an.
CONF_WITH_3D_OBJECTS 1                            # [GRAFIK] [DESKTOP] Dreidimensionale AES-Bedienelemente wie in TOS 4; aktiviert die Darstellung erhabener/vertiefter Objekte.

CONF_WITH_68040_PMMU 0                            # Speicherverwaltungseinheit des 68040; 1 richtet Tabellen ein, etwa für FreeMiNT unter aranym-mmu ohne set_mmu.prg, hier aus.
CONF_WITH_ACSI 1                                  # Atari-Festplattenbus ACSI; aktiviert den Zugriff auf angeschlossene Massenspeicher.
CONF_WITH_ADVANCED_CPU 1                          # Motorola-Prozessoren 68010 bis 68060; aktiviert deren zusätzliche CPU-Unterstützung.
